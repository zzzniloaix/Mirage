# compare_ladder_records.py:
"""Compares the ladder reports of a pool_check.py run with those of a committed record of the same
check: for each run directory (jobs-*) both have, the ladder's master (its SHA-256), the reference,
and each pair, by its index: the distorted playlist, its frames (every field of every frame, as
they are, scores at full precision) and its pooled values (those both reports have; a pooled
value that only one has is listed).

usage: python3 -I compare_ladder_records.py <new_dir> <committed_dir>
"""
import json
import lzma
import os
import sys

new_dir, old_dir = sys.argv[1], sys.argv[2]


def load(d, run):
    for name, op in (("report.json.xz", lzma.open), ("report.json", open)):
        p = os.path.join(d, run, name)
        if os.path.exists(p):
            with op(p, "rt", encoding="utf-8") as f:
                return json.load(f)
    return None


runs = sorted(r for r in os.listdir(new_dir)
              if r.startswith("jobs-") and os.path.isdir(os.path.join(old_dir, r)))
total_pairs = total_frames = differing = 0
for run in runs:
    a, b = load(new_dir, run), load(old_dir, run)
    if a is None or b is None:
        print(f"{run}: report missing ({'new' if a is None else 'committed'})")
        differing += 1
        continue
    notes = []
    if a["ladder"].get("sha256") != b["ladder"].get("sha256"):
        notes.append("master's SHA-256 differs")
    if a.get("reference") != b.get("reference"):
        notes.append("reference differs")
    pa = {p["index"]: p for p in a["pairs"]}
    pb = {p["index"]: p for p in b["pairs"]}
    if sorted(pa) != sorted(pb):
        notes.append(f"pairs differ: {sorted(pa)} / {sorted(pb)}")
    only = set()
    for i in sorted(set(pa) & set(pb)):
        x, y = pa[i], pb[i]
        total_pairs += 1
        total_frames += len(y.get("frames") or [])
        if x["dis"]["url"] != y["dis"]["url"]:
            notes.append(f"pair {i}: distorted {x['dis']['url']} / {y['dis']['url']}")
        fx, fy = x.get("frames") or [], y.get("frames") or []
        d = sum(1 for u, v in zip(fx, fy) if u != v) + abs(len(fx) - len(fy))
        if d:
            notes.append(f"pair {i}: {d} of {len(fy)} frames differ (new {len(fx)})")
            differing += d
        qx, qy = x.get("pooled") or {}, y.get("pooled") or {}
        for k in sorted(set(qx) & set(qy)):
            if qx[k] != qy[k]:
                notes.append(f"pair {i}: pooled {k} {qx[k]!r} / {qy[k]!r}")
                differing += 1
        only |= {f"{k} (new only)" for k in set(qx) - set(qy)}
        only |= {f"{k} (committed only)" for k in set(qy) - set(qx)}
    print(f"{run}: {len(set(pa) & set(pb))} pairs compared; "
          + ("; ".join(notes) if notes else "frames and pooled values the same")
          + (f"; pooled values in one report only: {', '.join(sorted(only))}" if only else ""))
print(f"in all: {len(runs)} runs, {total_pairs} pairs, {total_frames} committed frames, "
      f"{differing} differing")
