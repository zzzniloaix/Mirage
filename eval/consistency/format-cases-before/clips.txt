# The clips of eval/consistency/color_cases.py --set formats, made in <clips_dir>: for each, the command that made it, its stream as <ffprobe> reads it back, and the MD5 of its decoded frames.

## ref_ffv1-yuv420p10le_bt709-tv.mkv
$ <ffmpeg> -hide_banner -nostdin -v error -f lavfi -i testsrc2=size=1920x1080:rate=30000/1001 -frames:v 10 -vf format=yuv420p10le,setparams=colorspace=bt709:range=tv -c:v ffv1 -f matroska <clips_dir>/ref_ffv1-yuv420p10le_bt709-tv.mkv
$ <ffprobe> -v error -select_streams v:0 -show_entries stream=width,height,pix_fmt,color_space,color_range,color_primaries,color_transfer,chroma_location,field_order -of default=noprint_wrappers=1 <clips_dir>/ref_ffv1-yuv420p10le_bt709-tv.mkv
width=1920
height=1080
pix_fmt=yuv420p10le
color_range=tv
color_space=bt709
color_transfer=unknown
color_primaries=unknown
chroma_location=unspecified
field_order=progressive
$ <ffmpeg> -hide_banner -nostdin -v error -i <clips_dir>/ref_ffv1-yuv420p10le_bt709-tv.mkv -map 0:v:0 -c:v rawvideo -f md5 -
MD5=0307aa275a0dd75d9f1b98e56ec7d35b
