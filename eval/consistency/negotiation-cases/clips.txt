# The clips of eval/consistency/color_cases.py --set negotiation, made in <clips_dir>: for each, the command that made it, its stream as <ffprobe> reads it back, and the MD5 of its decoded frames.

## ref_libx264-yuvj420p_bt709-pc.mkv
$ <ffmpeg> -hide_banner -nostdin -v error -f lavfi -i testsrc2=size=1920x1080:rate=30000/1001 -frames:v 10 -vf format=yuv420p,setparams=colorspace=bt709:range=pc -c:v libx264 -qp 0 -f matroska <clips_dir>/ref_libx264-yuvj420p_bt709-pc.mkv
$ <ffprobe> -v error -select_streams v:0 -show_entries stream=width,height,pix_fmt,color_space,color_range,color_primaries,color_transfer,chroma_location,field_order -of default=noprint_wrappers=1 <clips_dir>/ref_libx264-yuvj420p_bt709-pc.mkv
width=1920
height=1080
pix_fmt=yuvj420p
color_range=pc
color_space=bt709
color_transfer=unknown
color_primaries=unknown
chroma_location=left
field_order=progressive
$ <ffmpeg> -hide_banner -nostdin -v error -i <clips_dir>/ref_libx264-yuvj420p_bt709-pc.mkv -map 0:v:0 -c:v rawvideo -f md5 -
MD5=d4a3f7e48e1e38381a738a75240e9844
